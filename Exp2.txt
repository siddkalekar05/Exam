//Installation Steps 

Step 1: Open Ubuntu WSL from Windows Terminal or PowerShell.
wsl

Step 2: Move to the NS-3 directory.
cd ~/ns-3-dev

Step 3: Create a new C++ file inside the scratch directory.
nano scratch/exp2-point-to-point.cc

Step 4: Write or paste the Practical 2 program from the lab manual. 

Step 5: Save the file by pressing Ctrl + O, then Enter. Exit using Ctrl + X.

Step 6: Run the Practical 2 simulation.
./ns3 run scratch/exp2-point-to-point

Step 7: Check the generated PCAP files.
ls *.pcap

Step 8: Open Wireshark.
wireshark

Step 9: Open the generated point-to-point PCAP file in Wireshark and apply the UDP display filter.
udp

Step 10: Observe 

/// Program 
#include "ns3/core-module.h" 
#include "ns3/network-module.h" 
#include "ns3/internet-module.h" 
#include "ns3/point-to-point-module.h" 
#include "ns3/applications-module.h" 
 
using namespace ns3; 
 
int main (int argc, char *argv[]) 
{ 
  NodeContainer nodes; 
  nodes.Create (2); 
 
  PointToPointHelper pointToPoint; 
  pointToPoint.SetDeviceAttribute ("DataRate", StringValue ("5Mbps")); 
  pointToPoint.SetChannelAttribute ("Delay", StringValue ("2ms")); 
 
  NetDeviceContainer devices = pointToPoint.Install (nodes); 
 
  InternetStackHelper stack; 
  stack.Install (nodes); 
 
  Ipv4AddressHelper address; 
  address.SetBase ("10.1.1.0", "255.255.255.0"); 
  Ipv4InterfaceContainer interfaces = address.Assign (devices); 
 
  UdpEchoServerHelper echoServer (9); 
  ApplicationContainer serverApps = echoServer.Install (nodes.Get (1)); 
  serverApps.Start (Seconds (1.0)); 
  serverApps.Stop (Seconds (5.0)); 
 
  UdpEchoClientHelper echoClient (interfaces.GetAddress (1), 9); 
  echoClient.SetAttribute ("MaxPackets", UintegerValue (1)); 
  echoClient.SetAttribute ("Interval", TimeValue (Seconds (1.0))); 
  echoClient.SetAttribute ("PacketSize", UintegerValue (1024)); 
 
  ApplicationContainer clientApps = echoClient.Install (nodes.Get (0)); 
  clientApps.Start (Seconds (2.0)); 
  clientApps.Stop (Seconds (5.0)); 
 
  pointToPoint.EnablePcapAll ("point-to-point-echo"); 
 
  Simulator::Run (); 
  Simulator::Destroy (); 
 
  return 0; 
}
