//Program

#include "ns3/core-module.h" 
#include "ns3/network-module.h" 
#include "ns3/internet-module.h" 
#include "ns3/point-to-point-module.h" 
#include "ns3/csma-module.h" 
#include "ns3/applications-module.h" 
 
using namespace ns3; 
 
int main (int argc, char *argv[]) 
{ 
  NodeContainer csmaNodes; 
  csmaNodes.Create (3); 
 
  NodeContainer p2pNodes; 
  p2pNodes.Add (csmaNodes.Get (2)); 
  Ptr<Node> remote = CreateObject<Node> (); 
  p2pNodes.Add (remote); 
 
  PointToPointHelper p2p; 
  p2p.SetDeviceAttribute ("DataRate", StringValue ("5Mbps")); 
  p2p.SetChannelAttribute ("Delay", StringValue ("2ms")); 
  NetDeviceContainer p2pDevices = p2p.Install (p2pNodes); 
 
  CsmaHelper csma; 
  csma.SetChannelAttribute ("DataRate", StringValue ("100Mbps")); 
  csma.SetChannelAttribute ("Delay", TimeValue (NanoSeconds (6560))); 
  NetDeviceContainer csmaDevices = csma.Install (csmaNodes); 
 
  InternetStackHelper stack; 
  stack.Install (csmaNodes); 
  stack.Install (remote); 
 
  Ipv4AddressHelper p2pAddress; 
  p2pAddress.SetBase ("10.1.2.0", "255.255.255.0"); 
  Ipv4InterfaceContainer p2pInterfaces = p2pAddress.Assign (p2pDevices); 
 
  Ipv4AddressHelper csmaAddress; 
  csmaAddress.SetBase ("10.1.1.0", "255.255.255.0"); 
  Ipv4InterfaceContainer csmaInterfaces = csmaAddress.Assign (csmaDevices); 
                                                                                                                                                                                                   
 
 
  UdpEchoServerHelper server (9); 
  ApplicationContainer serverApps = server.Install (remote); 
  serverApps.Start (Seconds (1.0)); 
  serverApps.Stop (Seconds (10.0)); 
 
  UdpEchoClientHelper client (p2pInterfaces.GetAddress (1), 9); 
  client.SetAttribute ("MaxPackets", UintegerValue (3)); 
  client.SetAttribute ("Interval", TimeValue (Seconds (1.0))); 
  client.SetAttribute ("PacketSize", UintegerValue (512)); 
 
  ApplicationContainer clientApps = client.Install (csmaNodes.Get (0)); 
  clientApps.Start (Seconds (2.0)); 
  clientApps.Stop (Seconds (8.0)); 
 
  p2p.EnablePcapAll ("p2p-csma"); 
  csma.EnablePcapAll ("csma"); 
 
  Simulator::Run (); 
  Simulator::Destroy (); 
  return 0; 
}
