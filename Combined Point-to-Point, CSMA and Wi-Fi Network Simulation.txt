//Execution Procedure

Step 1: Open Ubuntu WSL.
wsl

Step 2: Move to the NS-3 directory.
cd ~/ns-3-dev

Step 3: Create a new C++ file inside the scratch directory.
nano scratch/exp4-wifi.cc

Step 4: Write or paste the Practical 4 program from the lab manual.

Step 5: Save the file by pressing Ctrl + O, then Enter. Exit using Ctrl + X.

Step 6: Run the Practical 4 simulation.
./ns3 run scratch/exp4-wifi

Step 7: Check the generated PCAP files.
ls *.pcap

Step 8: Open Wireshark.
wireshark

Step 9: Open the generated Wi-Fi, CSMA or point-to-point PCAP files.

Step 10: Apply the UDP display filter to observe UDP packets.
udp

Step 11: Observe

//Program

#include "ns3/core-module.h" 
#include "ns3/network-module.h" 
#include "ns3/internet-module.h" 
#include "ns3/point-to-point-module.h" 
#include "ns3/csma-module.h" 
#include "ns3/wifi-module.h" 
#include "ns3/mobility-module.h" 
#include "ns3/applications-module.h" 
 
using namespace ns3; 
 
int main (int argc, char *argv[]) 
{ 
  // ---------- Nodes ---------- 
  NodeContainer p2pNodes; 
  p2pNodes.Create (2); 
 
  NodeContainer csmaNodes; 
  csmaNodes.Create (2); 
 
  NodeContainer wifiStaNodes; 
  wifiStaNodes.Create (2); 
 
  NodeContainer wifiApNode; 
  wifiApNode.Create (1); 
 
  // ---------- Point-to-Point ---------- 
  PointToPointHelper p2p; 
  p2p.SetDeviceAttribute ("DataRate", StringValue ("5Mbps")); 
  p2p.SetChannelAttribute ("Delay", StringValue ("2ms")); 
  NetDeviceContainer p2pDevices = p2p.Install (p2pNodes); 
 
  // ---------- CSMA ---------- 
  CsmaHelper csma; 
  csma.SetChannelAttribute ("DataRate", StringValue ("100Mbps")); 
  csma.SetChannelAttribute ("Delay", TimeValue (NanoSeconds (6560))); 
NetDeviceContainer csmaDevices = csma.Install (csmaNodes); 
// ---------- Wi-Fi ---------- 
WifiHelper wifi; 
wifi.SetStandard (WIFI_STANDARD_80211g); 
WifiMacHelper wifiMac; 
YansWifiChannelHelper channel = YansWifiChannelHelper::Default (); 
YansWifiPhyHelper phy; 
phy.SetChannel (channel.Create ()); 
Ssid ssid = Ssid ("ns3-wifi-network"); 
wifiMac.SetType ("ns3::StaWifiMac", 
"Ssid", SsidValue (ssid), 
"ActiveProbing", BooleanValue (false)); 
NetDeviceContainer staDevices = wifi.Install (phy, wifiMac, wifiStaNodes); 
// ---------- IP Addresses ---------- 
Ipv4AddressHelper p2pAddress; 
wifiMac.SetType ("ns3::ApWifiMac", 
"Ssid", SsidValue (ssid)); 
NetDeviceContainer apDevice = wifi.Install (phy, wifiMac, wifiApNode); 
// ---------- Mobility ---------- 
MobilityHelper mobility; 
mobility.SetMobilityModel ("ns3::ConstantPositionMobilityModel"); 
mobility.Install (wifiStaNodes); 
mobility.Install (wifiApNode); 
// ---------- Internet Stack ---------- 
InternetStackHelper stack; 
stack.Install (p2pNodes); 
stack.Install (csmaNodes); 
stack.Install (wifiStaNodes); 
stack.Install (wifiApNode); 
                                                                                                                                                                                                   
 
  p2pAddress.SetBase ("10.1.1.0", "255.255.255.0"); 
  p2pAddress.Assign (p2pDevices); 
 
  Ipv4AddressHelper csmaAddress; 
  csmaAddress.SetBase ("10.1.2.0", "255.255.255.0"); 
  csmaAddress.Assign (csmaDevices); 
 
  Ipv4AddressHelper wifiAddress; 
  wifiAddress.SetBase ("10.1.3.0", "255.255.255.0"); 
  Ipv4InterfaceContainer staInterfaces = wifiAddress.Assign (staDevices); 
  wifiAddress.Assign (apDevice); 
 
  // ---------- UDP Echo on Wi-Fi STA 1 ---------- 
  UdpEchoServerHelper server (9); 
  ApplicationContainer serverApps = server.Install (wifiStaNodes.Get (1)); 
  serverApps.Start (Seconds (1.0)); 
  serverApps.Stop (Seconds (10.0)); 
 
  UdpEchoClientHelper client (staInterfaces.GetAddress (1), 9); 
  client.SetAttribute ("MaxPackets", UintegerValue (3)); 
  client.SetAttribute ("Interval", TimeValue (Seconds (1.0))); 
  client.SetAttribute ("PacketSize", UintegerValue (512)); 
 
  ApplicationContainer clientApps = client.Install (wifiStaNodes.Get (0)); 
  clientApps.Start (Seconds (2.0)); 
  clientApps.Stop (Seconds (8.0)); 
 
  // ---------- Tracing ---------- 
  p2p.EnablePcapAll ("heterogeneous-p2p"); 
  csma.EnablePcapAll ("heterogeneous-csma"); 
  phy.EnablePcap ("heterogeneous-wifi", apDevice.Get (0), true); 
 
  Simulator::Run (); 
  Simulator::Destroy (); 
 
  return 0; 
}
